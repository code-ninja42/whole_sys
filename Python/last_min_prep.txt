class Parent:
    
    def __init__(self):
        self.name = "Rahi"
    
    def show(self):
        print("Parent Class")


class Child(Parent):
    
    def __init__(self):
        super().__init__()
        self.age = 20
    
    def display(self):
        super().show()
        print("Name :", self.name)
        print("Age :", self.age)


c = Child()

c.display()
